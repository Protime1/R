# ---------- Windows DPAPI ----------

class DATA_BLOB(ctypes.Structure):
    _fields_ = [
        ("cbData", wintypes.DWORD),
        ("pbData", ctypes.POINTER(ctypes.c_char)),
    ]


def _blob_to_bytes(blob):
    size = blob.cbData
    data = ctypes.string_at(blob.pbData, size)
    ctypes.windll.kernel32.LocalFree(blob.pbData)
    return data


def _bytes_to_blob(data):
    buf = ctypes.create_string_buffer(data, len(data))
    blob = DATA_BLOB()
    blob.cbData = len(data)
    blob.pbData = ctypes.cast(buf, ctypes.POINTER(ctypes.c_char))
    return blob, buf


def dpapi_encrypt(plaintext):
    try:
        data = plaintext.encode("utf-8")
        blob_in, _buf = _bytes_to_blob(data)
        blob_out = DATA_BLOB()
        ok = ctypes.windll.crypt32.CryptProtectData(
            ctypes.byref(blob_in),
            u"LunaStoreXP",
            None, None, None, 0,
            ctypes.byref(blob_out),
        )
        if not ok:
            return None
        return _blob_to_bytes(blob_out)
    except Exception:
        return None


def dpapi_decrypt(ciphertext):
    try:
        blob_in, _buf = _bytes_to_blob(ciphertext)
        blob_out = DATA_BLOB()
        ok = ctypes.windll.crypt32.CryptUnprotectData(
            ctypes.byref(blob_in),
            None, None, None, None, 0,
            ctypes.byref(blob_out),
        )
        if not ok:
            return None
        raw = _blob_to_bytes(blob_out)
        return raw.decode("utf-8")
    except Exception:
        return None


# ---------- TLS-диагностика ----------

def check_tls_support(base_url):
    """Проверяет HTTPS. Возвращает (ok, message)."""
    try:
        resp = requests.get(base_url, timeout=10, verify=False)
        return True, "TLS OK (HTTP %d)" % resp.status_code
    except requests.exceptions.SSLError as e:
        return False, "SSL: %s" % str(e)[:120]
    except requests.exceptions.RequestException as e:
        return False, "Сеть: %s" % str(e)[:120]

    # ==================== Старт ====================

    def _initial_checks(self):
        base = self.instance_url.get().strip().rstrip("/")
        ok, msg = check_tls_support(base)
        if not ok:
            messagebox.showwarning(
                "Проблема с соединением",
                "Не удалось установить HTTPS-соединение:\n\n"
                "%s\n\n"
                "Возможные причины:\n"
                "1. На этой машине отключён TLS 1.2.\n"
                "2. Прокси или файрвол блокирует соединение.\n"
                "3. Устаревшие сертификаты Windows XP.\n\n"
                "Установите обновление KB948963 или обновите корневые сертификаты." % msg
            )
            self._set_status("TLS недоступен", self.COLORS["danger"])
        self._try_restore_session()

    # ==================== Сессия (DPAPI) ====================

    def _save_session(self):
        if not self.refresh_token:
            return
        try:
            if not os.path.isdir(APP_DIR):
                os.makedirs(APP_DIR)

            payload = {
                "instance_url": self.instance_url.get(),
                "username": self.username.get(),
                "refresh": self.refresh_token,
            }
            blob = dpapi_encrypt(json.dumps(payload))
            if blob is None:
                self._set_status("Не удалось зашифровать сессию", self.COLORS["warn"])
                return

            with open(SESSION_FILE, "wb") as f:
                f.write(blob)
        except Exception:
            pass

    def _clear_session(self):
        try:
            if os.path.isfile(SESSION_FILE):
                os.remove(SESSION_FILE)
        except Exception:
            pass

    def _try_restore_session(self):
        old_file = os.path.join(APP_DIR, "session.json")
        if os.path.isfile(old_file):
            try:
                os.remove(old_file)
            except Exception:
                pass

        if not os.path.isfile(SESSION_FILE):
            return
        try:
            with open(SESSION_FILE, "rb") as f:
                blob = f.read()

            plaintext = dpapi_decrypt(blob)
            if plaintext is None:
                self._clear_session()
                self._set_status("Сохранённая сессия недействительна")
                return

            data = json.loads(plaintext)
            self.instance_url.set(data.get("instance_url", self.instance_url.get()))
            self.username.set(data.get("username", ""))
            self.refresh_token = data.get("refresh")

            if self.refresh_token:
                self._set_status("Восстановление сессии...", self.COLORS["accent"])
                self._refresh_access_token(then=self._on_session_restored)
        except Exception:
            self._clear_session()

    def _on_session_restored(self, ok):
        if ok:
            self.login_btn.state(["disabled"])
            self.logout_btn.state(["!disabled"])
            self.refresh_btn.state(["!disabled"])
            self.search_btn.state(["!disabled"])
            self.clear_btn.state(["!disabled"])
            self._set_status("Сессия восстановлена", self.COLORS["accent2"])
            self._load_categories()
            self._load_apps()
        else:
            self._clear_session()
            self._set_status("Не удалось восстановить сессию")

    # ==================== Авторизация ====================

    def _do_login(self):
        url = self._api_url("/method/v2/auth/token/")
        self._set_status("Вход...", self.COLORS["accent"])
        self.login_btn.state(["disabled"])

        def task():
            try:
                resp = requests.post(
                    url,
                    json={"username": self.username.get(),
                          "password": self.password.get()},
                    timeout=15, verify=False
                )
                data = resp.json()
                if "access" in data:
                    self.token = data["access"]
                    self.refresh_token = data.get("refresh")
                    self.root.after(0, self._on_login_success)
                else:
                    code = data.get("code", 0)
                    msg = data.get("message", "Ошибка входа")
                    self.root.after(0, lambda: self._on_login_error(code, msg))
            except Exception as e:
                self.root.after(0, lambda: self._on_login_error(0, str(e)))

        threading.Thread(target=task, daemon=True).start()

    def _on_login_success(self):
        self._set_status("Вход выполнен", self.COLORS["accent2"])
        self.login_btn.state(["disabled"])
        self.logout_btn.state(["!disabled"])
        self.refresh_btn.state(["!disabled"])
        self.search_btn.state(["!disabled"])
        self.clear_btn.state(["!disabled"])
        self._save_session()
        self._load_categories()
        self._load_apps()

    def _on_login_error(self, code, msg):
        self.login_btn.state(["!disabled"])
        self._set_status("Ошибка входа", self.COLORS["danger"])
        errors = {
            1000: "Пользователь не найден",
            1001: "Пользователь заблокирован",
            4000: "Требуется 2FA (не поддерживается)",
            4001: "Неверный код 2FA",
        }
        messagebox.showerror("Ошибка входа", errors.get(code, "Код %d: %s" % (code, msg)))

    def _do_logout(self):
        self.token = None

    # ==================== Автообновление токена ====================

    def _refresh_access_token(self, then=None):
        url = self._api_url("/method/v2/auth/token/refresh/")

        def task():
            with self._refresh_lock:
                ok = False
                try:
                    resp = requests.post(
                        url, json={"refresh": self.refresh_token},
                        timeout=15, verify=False
                    )
                    if resp.status_code == 200:
                        data = resp.json()
                        if "access" in data:
                            self.token = data["access"]
                            if "refresh" in data:
                                self.refresh_token = data["refresh"]
                                self._save_session()
                            ok = True
                except Exception:
                    ok = False
                if then:
                    self.root.after(0, lambda: then(ok))

        threading.Thread(target=task, daemon=True).start()

    def _api_request_with_retry(self, method, url, **kwargs):
        headers = kwargs.pop("headers", {}) or {}
        if self.token:
            headers["Authorization"] = "Bearer " + self.token

        try:
            resp = requests.request(method, url, headers=headers, verify=False, **kwargs)
        except Exception:
            return None

        if resp.status_code == 401 and self.refresh_token:
            result = [False]
            done = threading.Event()

            def cb(ok):
                result[0] = ok
                done.set()

            self._refresh_access_token_sync(cb)
            done.wait(timeout=20)

            if result[0]:
                headers["Authorization"] = "Bearer " + self.token
                try:
                    resp = requests.request(
                        method, url, headers=headers, verify=False, **kwargs
                    )
                except Exception:
                    return None

        return resp

    def _refresh_access_token_sync(self, callback):
        url = self._api_url("/method/v2/auth/token/refresh/")
        try:
            with self._refresh_lock:
                resp = requests.post(
                    url, json={"refresh": self.refresh_token},
                    timeout=15, verify=False
                )
                if resp.status_code == 200:
                    data = resp.json()
                    if "access" in data:
                        self.token = data["access"]
                        if "refresh" in data:
                            self.refresh_token = data["refresh"]
                            self._save_session()
                        callback(True)
                        return
        except Exception:
            pass
        callback(False)

    def _load_apps(self, append=False):
        if not self.token:
            return
        if append and self.is_loading:
            return

        # Лимит на количество приложений
        if not append and self.offset + self.limit > MAX_APPS:
            self._set_status("Достигнут лимит. Используйте поиск.", self.COLORS["warn"])
            self.load_more_btn.state(["disabled"])
            return
        if append and len(self.apps_data) >= MAX_APPS:
            self._set_status("Достигнут лимит %d приложений." % MAX_APPS, self.COLORS["warn"])
            self.load_more_btn.state(["disabled"])
            return

        self.is_loading = True
        self._load_generation += 1
        my_gen = self._load_generation

        self.refresh_btn.state(["disabled"])
        self.load_more_btn.state(["disabled"])

        if not append:
            self.offset = 0
            self._set_status("Загрузка приложений...", self.COLORS["accent"])

        url = self._api_url("/method/v2/marketplace/")
        params = {"limit": self.limit, "offset": self.offset}
        if self.active_category_id is not None:
            params["category"] = self.active_category_id

        def task():
            resp = self._api_request_with_retry("GET", url, params=params, timeout=20)
            if my_gen != self._load_generation:
                return
            if resp is None:
                self.root.after(0, lambda: self._set_status(
                    "Нет соединения", self.COLORS["danger"]
                ))
                return
            if resp.status_code == 401:
                self.root.after(0, self._on_token_expired)
                return
            try:
                data = resp.json()
                results = data.get("results", [])
                self.root.after(0, lambda: self._on_apps_loaded(results, append, my_gen))
            except Exception as e:
                self.root.after(0, lambda: self._set_status(
                    "Ошибка: " + str(e)[:60], self.COLORS["danger"]
                ))

        threading.Thread(target=task, daemon=True).start()

    def _on_apps_loaded(self, results, append, generation=0):
        if generation and generation != self._load_generation:
            return

        self.is_loading = False
        self.refresh_btn.state(["!disabled"])

        if not append:
            self._clear_list()
            self.apps_data = []

        if results:
            start_index = len(self.apps_data)
            # Ограничиваем общее количество
            remaining = MAX_APPS - start_index
            if remaining <= 0:
                self.load_more_btn.state(["disabled"])
                return
            results = results[:remaining]

            self.apps_data.extend(results)
            for i, app in enumerate(results):
                self._render_row(start_index + i, app)

            self.offset += len(results)
            self._set_status("Загружено приложений: %d" % len(self.apps_data))

            if len(self.apps_data) >= MAX_APPS:
                self.load_more_btn.state(["disabled"])
                self._set_status("Достигнут лимит %d" % MAX_APPS, self.COLORS["warn"])
            else:
                self.load_more_btn.state(["!disabled"])
        else:
            if not append:
                empty = tk.Label(
                    self.list_inner, text="Приложений нет",
                    bg=self.COLORS["surface2"], fg=self.COLORS["subtext"],
                    font=("Arial", 10), anchor=tk.W
                )
                empty.pack(fill=tk.X, padx=8, pady=8)
                self._row_widgets.append(empty)
            self.load_more_btn.state(["disabled"])
            self._set_status("Больше приложений нет")

    def _search_apps(self, query):
        if not self.token:
            return
        self.is_loading = True
        self._load_generation += 1
        my_gen = self._load_generation

        self.refresh_btn.state(["disabled"])
        self.search_btn.state(["disabled"])
        self.load_more_btn.state(["disabled"])
        self.offset = 0
        self._set_status("Поиск: " + query, self.COLORS["accent"])

        url = self._api_url("/method/v2/marketplace/search/")
        params = {"query": query, "limit": self.limit, "offset": 0}
        if self.active_category_id is not None:
            params["category"] = self.active_category_id

        def task():
            resp = self._api_request_with_retry("GET", url, params=params, timeout=20)
            if my_gen != self._load_generation:
                return
            if resp is None:
                self.root.after(0, lambda: self._set_status(
                    "Нет соединения", self.COLORS["danger"]
                ))
                return
            if resp.status_code == 401:
                self.root.after(0, self._on_token_expired)
                return
            try:
                data = resp.json()
                results = data.get("results", data.get("hits", []))
                self.root.after(0, lambda: self._on_search_done(results, query, my_gen))
            except Exception as e:
                self.root.after(0, lambda: self._set_status(
                    "Ошибка поиска: " + str(e)[:60], self.COLORS["danger"]
                ))

        threading.Thread(target=task, daemon=True).start()

    def _on_search_done(self, results, query, generation=0):
        if generation and generation != self._load_generation:
            return

        self.is_loading = False
        self.search_btn.state(["!disabled"])
        self.refresh_btn.state(["!disabled"])
        self.load_more_btn.state(["disabled"])

        self._clear_list()
        self.apps_data = results[:MAX_APPS]
        self.offset = len(self.apps_data)

        if not self.apps_data:
            empty = tk.Label(
                self.list_inner, text="Ничего не найдено: " + query,
                bg=self.COLORS["surface2"], fg=self.COLORS["subtext"],
                font=("Arial", 10), anchor=tk.W
            )
            empty.pack(fill=tk.X, padx=8, pady=8)
            self._row_widgets.append(empty)
            self._set_status("Ничего не найдено")
            return

        for i, app in enumerate(self.apps_data):
            self._render_row(i, app)

        self._set_status("Найдено: %d" % len(self.apps_data), self.COLORS["accent2"])

    def _on_token_expired(self):
        self._set_status("Сессия истекла, войдите заново", self.COLORS["danger"])
        self._clear_session()
        self._do_logout()

    def _load_more(self):

    def _icon_cache_path(self, app_id):
        return os.path.join(CACHE_DIR, "app_%s.png" % app_id)

    def _get_icon_for_app(self, app):
        if not HAS_PIL:
            return None

        app_id = app.get("id")
        if not app_id:
            return None

        icon_url = (
            app.get("icon") or app.get("logo") or
            app.get("image") or app.get("icon_url") or ""
        )
        if not icon_url:
            return None

        if icon_url.startswith("/"):
            icon_url = self.instance_url.get().strip().rstrip("/") + icon_url

        cache_path = self._icon_cache_path(app_id)

        try:
            if not os.path.isfile(cache_path):
                if not os.path.isdir(CACHE_DIR):
                    os.makedirs(CACHE_DIR)
                resp = requests.get(icon_url, timeout=10, verify=False)
                if resp.status_code != 200:
                    return None
                with open(cache_path, "wb") as f:
                    f.write(resp.content)

            img = Image.open(cache_path)
            img = img.convert("RGBA")
            img.thumbnail(ICON_SIZE, Image.LANCZOS)
            return ImageTk.PhotoImage(img)
        except Exception:
            return None

    def _hover_row(self, row, entering):
        c = self.COLORS
        if self._selected_index is not None:
            try:
                if row == self._row_widgets[self._selected_index]:
                    return
            except IndexError:
                pass
        color = c["surface"] if entering else c["surface2"]
        try:
            row.configure(bg=color)
            for w in row.winfo_children():
                w.configure(bg=color)
        except Exception:
            pass

    def _select_row(self, index):
        c = self.COLORS
        if self._selected_index is not None:
            try:
                old = self._row_widgets[self._selected_index]
                old.configure(bg=c["surface2"])
                for w in old.winfo_children():
                    w.configure(bg=c["surface2"])
            except Exception:
                pass

        self._selected_index = index

        try:
            row = self._row_widgets[index]
            row.configure(bg=c["accent"])
            for w in row.winfo_children():
                w.configure(bg=c["accent"])
        except Exception:
            pass

            self._show_app_details(self.apps_data[index])

    # ==================== Скачивание ====================

    def _download_app(self):
        if not self.selected_app:
            return
        app_id = self.selected_app.get("id")
        if not app_id:
            messagebox.showwarning("Внимание", "У приложения нет ID")
            return

        url = self._api_url("/method/v2/distribution/")
        params = {"application": app_id}
        self._set_status("Получение ссылки...", self.COLORS["accent"])

        def task():
            resp = self._api_request_with_retry("GET", url, params=params, timeout=15)
            if resp is None:
                self.root.after(0, lambda: messagebox.showerror("Ошибка", "Нет соединения"))
                return
            try:
                data = resp.json()
                results = data.get("results", [])
                if not results:
                    self.root.after(0, lambda: messagebox.showinfo("Инфо", "Нет файлов"))
                    return
                dist = results[0]
                link = dist.get("file") or dist.get("url") or ""
                if not link:
                    self.root.after(0, lambda: messagebox.showinfo("Инфо", "Нет ссылки"))
                    return
                self.root.after(0, lambda: self._start_file_download(link, dist))
            except Exception as e:
                self.root.after(0, lambda: messagebox.showerror("Ошибка", str(e)[:100]))

        threading.Thread(target=task, daemon=True).start()

    def _start_file_download(self, url, dist_info):
        default_name = dist_info.get("filename") or url.split("/")[-1] or "download.bin"
        if "?" in default_name:
            default_name = default_name.split("?")[0]
        if not default_name:
            default_name = "lunastore_download.bin"

        initial_dir = self.settings.get("download_dir", os.path.expanduser("~"))

        path = filedialog.asksaveasfilename(
            title="Сохранить как",
            initialdir=initial_dir,
            initialfile=default_name
        )
        if not path:
            return

        # Запоминаем папку
        self.settings["download_dir"] = os.path.dirname(path)
        self._save_settings_file()

        prog = tk.Toplevel(self.root)
        prog.title("Скачивание")
        prog.geometry("400x140")
        prog.configure(bg=self.COLORS["surface"])
        prog.transient(self.root)
        prog.grab_set()

        tk.Label(
            prog, text=default_name[:70],
            bg=self.COLORS["surface"], fg=self.COLORS["text"],
            font=("Arial", 10), wraplength=360
        ).pack(pady=(12, 6))

        bar = ttk.Progressbar(prog, mode="determinate", length=360, maximum=100)
        bar.pack(pady=6)

        info = tk.Label(
            prog, text="0%", bg=self.COLORS["surface"], fg=self.COLORS["subtext"],
            font=("Arial", 9)
        )
        info.pack()

        state = {"cancel": False}

        def on_cancel():
            state["cancel"] = True

        ttk.Button(prog, text="Отмена", style="Ghost.TButton",
                   command=on_cancel).pack(pady=6)

        def progress_task():
            try:
                r = requests.get(url, stream=True, timeout=30, verify=False)
                if r.status_code != 200:
                    r.close()
                    self.root.after(0, lambda: messagebox.showerror(
                        "Ошибка", "HTTP %d" % r.status_code
                    ))
                    self.root.after(0, prog.destroy)
                    return

                total = int(r.headers.get("content-length", 0))
                downloaded = 0
                chunk_size = 8192

                with open(path, "wb") as f:
                    for chunk in r.iter_content(chunk_size=chunk_size):
                        if state["cancel"]:
                            r.close()
                            try:
                                f.close()
                            except Exception:
                                pass
                            try:
                                os.remove(path)
                            except Exception:
                                pass
                            self.root.after(0, prog.destroy)
                            self.root.after(0, lambda: self._set_status("Скачивание отменено"))
                            return
                        if not chunk:
                            continue
                        f.write(chunk)
                        downloaded += len(chunk)
                        if total > 0:
                            pct = int(downloaded * 100 / total)
                            self.root.after(
                                0,
                                lambda p=pct, d=downloaded, t=total:
                                    self._update_progress(bar, info, p, d, t)
                            )
                        else:
                            self.root.after(
                                0,
                                lambda d=downloaded:
                                    self._update_progress(bar, info, 0, d, 0)
                            )

                r.close()
                self.root.after(0, lambda: self._download_done(prog, path, downloaded))
            except Exception as e:
                self.root.after(0, lambda: messagebox.showerror(
                    "Ошибка скачивания", str(e)[:150]
                ))
                self.root.after(0, prog.destroy)

        threading.Thread(target=progress_task, daemon=True).start()

    def _update_progress(self, bar, info_label, pct, downloaded, total):
        try:
            if total > 0:
                bar["value"] = pct
                mb_done = downloaded / (1024.0 * 1024.0)
                mb_total = total / (1024.0 * 1024.0)
                info_label.config(text="%d%%  (%.1f / %.1f МБ)" % (pct, mb_done, mb_total))
            else:
                mb_done = downloaded / (1024.0 * 1024.0)
                info_label.config(text="%.1f МБ" % mb_done)
        except Exception:
            pass

    def _download_done(self, prog_window, path, downloaded):
        try:
            prog_window.destroy()
        except Exception:
            pass
        self._set_status("Скачано: " + os.path.basename(path), self.COLORS["accent2"])
        if messagebox.askyesno("Готово", "Файл скачан:\n%s\n\nОткрыть папку?" % path):
            try:
                os.startfile(os.path.dirname(path))
            except Exception:
                pass

    def _get_big_icon(self, app, size):
        """Загружает иконку большего размера для карточки."""
        if not HAS_PIL:
            return None
        app_id = app.get("id")
        if not app_id:
            return None
        icon_url = (
            app.get("icon") or app.get("logo") or
            app.get("image") or app.get("icon_url") or ""
        )
        if not icon_url:
            return None
        if icon_url.startswith("/"):
            icon_url = self.instance_url.get().strip().rstrip("/") + icon_url

        # Используем тот же кэш, но пересобираем в большем размере
        cache_path = self._icon_cache_path(app_id)
        try:
            if not os.path.isfile(cache_path):
                return None
            img = Image.open(cache_path)
            img = img.convert("RGBA")
            img.thumbnail(size, Image.LANCZOS)
            return ImageTk.PhotoImage(img)
        except Exception:
            return None

    def _open_settings_window(self):
        c = self.COLORS
        win = tk.Toplevel(self.root)
        win.title("Настройки")
        win.geometry("480x320")
        win.configure(bg=c["surface"])
        win.transient(self.root)
        win.grab_set()

        tk.Label(
            win, text="Настройки", bg=c["surface"], fg=c["text"],
            font=("Arial", 14, "bold")
        ).pack(anchor=tk.W, padx=16, pady=(16, 12))

        # Папка загрузок
        tk.Label(win, text="Папка для скачивания",
                 bg=c["surface"], fg=c["subtext"], font=("Arial", 9)
                 ).pack(anchor=tk.W, padx=16)

        dir_frame = tk.Frame(win, bg=c["surface"])
        dir_frame.pack(fill=tk.X, padx=16, pady=(2, 12))

        dir_var = tk.StringVar(value=self.settings.get("download_dir", ""))
        ttk.Entry(dir_frame, textvariable=dir_var).pack(
            side=tk.LEFT, fill=tk.X, expand=True
        )

        def choose_dir():
            d = filedialog.askdirectory(
                title="Выберите папку",
                initialdir=dir_var.get() or os.path.expanduser("~")
            )
            if d:
                dir_var.set(d)

        ttk.Button(dir_frame, text="...", style="Ghost.TButton",
                   command=choose_dir, width=3).pack(side=tk.LEFT, padx=(4, 0))

        # Кнопки действий
        tk.Frame(win, bg=c["surface2"], height=1).pack(fill=tk.X, padx=16, pady=8)

        tk.Label(win, text="Действия",
                 bg=c["surface"], fg=c["subtext"], font=("Arial", 9)
                 ).pack(anchor=tk.W, padx=16)

        actions = tk.Frame(win, bg=c["surface"])
        actions.pack(fill=tk.X, padx=16, pady=(4, 12))

        def clear_icons():
            if not os.path.isdir(CACHE_DIR):
                messagebox.showinfo("Кэш", "Кэш иконок пуст")
                return
            count = 0
            for f in os.listdir(CACHE_DIR):
                try:
                    os.remove(os.path.join(CACHE_DIR, f))
                    count += 1
                except Exception:
                    pass
            self._icon_refs = {}
            messagebox.showinfo("Кэш", "Удалено файлов: %d" % count)

        def reset_session():
            if messagebox.askyesno("Сброс", "Удалить сохранённую сессию?"):
                self._clear_session()
                self.refresh_token = None
                self.token = None
                messagebox.showinfo("Сброс", "Сессия удалена. Перезайдите.")

        def clear_all():
            if messagebox.askyesno("Подтверждение",
                                   "Удалить ВСЕ данные клиента (сессию, настройки, кэш)?"):
                try:
                    for f in [SESSION_FILE, SETTINGS_FILE]:
                        if os.path.isfile(f):
                            os.remove(f)
                    if os.path.isdir(CACHE_DIR):
                        for f in os.listdir(CACHE_DIR):
                            try:
                                os.remove(os.path.join(CACHE_DIR, f))
                            except Exception:
                                pass
                except Exception:
                    pass
                messagebox.showinfo("Готово", "Данные удалены")

        ttk.Button(actions, text="Очистить кэш иконок", style="Ghost.TButton",
                   command=clear_icons).pack(side=tk.LEFT, padx=(0, 4))
        ttk.Button(actions, text="Сбросить сессию", style="Ghost.TButton",
                   command=reset_session).pack(side=tk.LEFT, padx=4)
        ttk.Button(actions, text="Удалить всё", style="Ghost.TButton",
                   command=clear_all).pack(side=tk.LEFT, padx=4)

        # Кнопки «Сохранить» / «Отмена»
        footer = tk.Frame(win, bg=c["surface"])
        footer.pack(fill=tk.X, side=tk.BOTTOM, padx=16, pady=12)

        def save_and_close():
            self.settings["download_dir"] = dir_var.get().strip() or os.path.expanduser("~")
            self._save_settings_file()
            win.destroy()

        ttk.Button(footer, text="Сохранить", style="Accent.TButton",
                   command=save_and_close).pack(side=tk.RIGHT, padx=4)
        ttk.Button(footer, text="Отмена", style="Ghost.TButton",
                   command=win.destroy).pack(side=tk.RIGHT, padx=4)
