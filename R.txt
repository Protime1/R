# ==================== RESOURCE PATH ====================
def resource_path(relative):
    """Путь к ресурсу - работает и в .py, и в .exe."""
    if hasattr(sys, '_MEIPASS'):
        return os.path.join(sys._MEIPASS, relative)
    return os.path.join(os.path.dirname(os.path.abspath(__file__)), relative)

# ==================== ВЕБ-СЕРВЕР ====================
def find_free_port(start=8080, end=8100):
    """Ищет свободный порт в диапазоне."""
    for port in range(start, end + 1):
        s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        try:
            s.bind(("127.0.0.1", port))
            s.close()
            return port
        except OSError:
            s.close()
            continue
    raise RuntimeError("Нет свободных портов в {0}-{1}".format(start, end))


class WebHandler(SimpleHTTPRequestHandler):
    def __init__(self, *args, **kwargs):
        templates = resource_path("templates")
        SimpleHTTPRequestHandler.__init__(self, *args, directory=templates, **kwargs)

    def log_message(self, fmt, *args):
        print("[WEB] {0} - {1}".format(self.address_string(), fmt % args))


def start_web_server():
    """Запускает веб-сервер и открывает браузер."""
    templates = resource_path("templates")

    if not os.path.exists(templates):
        print("Папка не найдена: {0}".format(templates))
        return

    index = os.path.join(templates, "index.html")
    if not os.path.exists(index):
        print("Файл не найден: {0}".format(index))
        return

    try:
        port = find_free_port()
    except RuntimeError as e:
        print("Ошибка: {0}".format(e))
        return

    print("[WEB] Запуск сервера на http://127.0.0.1:{0}/".format(port))
    print("[WEB] Директория: {0}".format(templates))

    try:
        server = HTTPServer(("127.0.0.1", port), WebHandler)
    except Exception as e:
        print("Не удалось создать сервер: {0}".format(e))
        traceback.print_exc()
        return

    url = "http://127.0.0.1:{0}/index.html".format(port)

    def open_browser():
        time.sleep(0.8)
        print("[WEB] Открываю браузер: {0}".format(url))
        webbrowser.open(url)

    t = threading.Thread(target=open_browser)
    t.daemon = True
    t.start()

    print("[WEB] Сервер готов.")
    try:
        server.serve_forever()
    except KeyboardInterrupt:
        print("\n[WEB] Сервер остановлен.")

### Если хотите, чтобы **видео работало и на IE8**:

Замените в `index.html` блок с `<video>` на:

```html
<div class="video-panel">
    <h3>Видео</h3>
    <p>Название видео</p>
    <p>
        <a href="URL_видео" target="_blank">
            Смотреть в Internet Archive (откроется в браузере) →
        </a>
    </p>
</div>
```

Тогда на XP будет **кнопка-ссылка**, которая откроет видео на сайте Archive.org.
