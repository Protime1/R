from PyQt5.QtCore import Qt
from PyQt5.QtWidgets import QApplication, QWidget, QLabel, QVBoxLayout, QRadioButton, QHBoxLayout, QGroupBox, QPushButton, QButtonGroup
from random import shuffle

app = QApplication([])
main_win = QWidget()
main_win.setWindowTitle("Memo card")
main_win.resize(450, 300)

MAINtext = QLabel()
MAINtext.setAlignment(Qt.AlignCenter)
MAINtext.setWordWrap(True)

RadioGroupBox = QGroupBox("Варианты:")
button1 = QRadioButton()
button2 = QRadioButton()
button3 = QRadioButton()
button4 = QRadioButton()

RadioGroup = QButtonGroup()
RadioGroup.addButton(button1)
RadioGroup.addButton(button2)
RadioGroup.addButton(button3)
RadioGroup.addButton(button4)

layout1 = QHBoxLayout()
col1 = QVBoxLayout()
col2 = QVBoxLayout()
col1.addWidget(button1)
col1.addWidget(button2)
col2.addWidget(button3)
col2.addWidget(button4)
layout1.addLayout(col1)
layout1.addLayout(col2)
RadioGroupBox.setLayout(layout1)

ansGroupBox = QGroupBox("Результат теста:")
winlabel = QLabel("")
winlabel.setAlignment(Qt.AlignCenter)
layout2 = QHBoxLayout()
layout2.addWidget(winlabel)
ansGroupBox.setLayout(layout2)
ansGroupBox.hide()

Otv_Button = QPushButton("Ответить")
Otv_Button.setMinimumHeight(35)

liner = QVBoxLayout()
liner.setContentsMargins(20, 20, 20, 20)
liner.setSpacing(15)
liner.addWidget(MAINtext, alignment=Qt.AlignCenter)
liner.addWidget(RadioGroupBox)
liner.addWidget(ansGroupBox)
liner.addWidget(Otv_Button)
main_win.setLayout(liner)

buttons = [button1, button2, button3, button4]
answer = buttons
idx = 0

def ask(question, right_answer, wrong1, wrong2, wrong3):
    shuffle(buttons)
    answer[0].setText(right_answer)
    answer[1].setText(wrong1)
    answer[2].setText(wrong2)
    answer[3].setText(wrong3)
    MAINtext.setText(question)
    RadioGroupBox.show()
    ansGroupBox.hide()
    Otv_Button.setText("Ответить")
    RadioGroup.setExclusive(False)
    for b in buttons:
        b.setChecked(False)
    RadioGroup.setExclusive(True)

def check_anw():
    if answer[0].isChecked():
        show_correct("Правильно!")
    elif RadioGroup.checkedButton():
        show_correct(f"Неверно! Правильный ответ: {answer[0].text()}")

def show_correct(res):
    winlabel.setText(res)
    RadioGroupBox.hide()
    ansGroupBox.show()
    Otv_Button.setText("Следующий вопрос")

def test_text():
    global idx
    if Otv_Button.text() == "Ответить":
        check_anw()
    else:
        idx += 1
        if idx == 1:
            ask("Государственный язык Бразилии", "Португальский", "Испанский", "Английский", "Итальянский")
        elif idx == 2:
            ask("Столица Франции", "Париж", "Лион", "Марсель", "Ницца")
        else:
            idx = 0
            ask("Какой национальности не существует?", "Смурфы", "Энцы", "Чулымцы", "Алеуты")

ask("Какой национальности не существует?", "Смурфы", "Энцы", "Чулымцы", "Алеуты")
Otv_Button.clicked.connect(test_text)
main_win.show()
app.exec_()
